def react_agent(question, tools, max_steps):
    tools={}
    history=[]
    for _ in range(max_steps):
        response=llm(
            f"""
            prompt:{question},
            context:{history}

            return:
            Thought:思考过程，
            Action:工具,
            Action_input:工具参数

            如果有final answer，直接返回
            """
        )
        if "Final answer" in response:
            return response.split("Final answer:")[-1].strip()
        
        action=parse_action(response)
        action_input=parse_action_input(response)

        ovservation=tools[action](action_input)

        history.append(
            "Thought":response,
            "Action":action,
            "observation":ovservation
        )

    return "No answer"
